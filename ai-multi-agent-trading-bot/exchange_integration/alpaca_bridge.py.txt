"""
Alpaca Bridge: Koneksi ke Alpaca API untuk trading

Fungsi:
1. Koneksi ke Alpaca API (Paper & Live)
2. Submit order (Buy/Sell)
3. Get account information
4. Get current prices
5. Get position information
"""

import os
import logging
import requests
from typing import Dict, List, Optional, Any
from datetime import datetime
import json

# Setup logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class AlpacaBridge:
    """
    Bridge untuk koneksi ke Alpaca API
    """
    
    def __init__(self, config: Dict = None):
        """
        Initialize Alpaca Bridge
        
        Args:
            config: Konfigurasi dengan API keys
        """
        self.config = config or {}
        
        # Get API keys from environment
        self.api_key = os.getenv('ALPACA_API_KEY', '')
        self.api_secret = os.getenv('ALPACA_SECRET_KEY', '')
        self.paper_mode = os.getenv('ALPACA_PAPER', 'True').lower() == 'true'
        
        # Set base URLs
        if self.paper_mode:
            self.base_url = 'https://paper-api.alpaca.markets'
            self.data_url = 'https://data.alpaca.markets'
        else:
            self.base_url = 'https://api.alpaca.markets'
            self.data_url = 'https://data.alpaca.markets'
        
        # Headers for API requests
        self.headers = {
            'APCA-API-KEY-ID': self.api_key,
            'APCA-API-SECRET-KEY': self.api_secret,
            'Content-Type': 'application/json'
        }
        
        # Check connection
        self.connected = self._check_connection()
        
        if self.connected:
            logger.info(f"Alpaca Bridge initialized in {'PAPER' if self.paper_mode else 'LIVE'} mode")
        else:
            logger.warning("Alpaca Bridge initialized but NOT connected - check API keys")
    
    def _check_connection(self) -> bool:
        """Check if connection to Alpaca is working"""
        try:
            response = requests.get(
                f"{self.base_url}/v2/account",
                headers=self.headers
            )
            return response.status_code == 200
        except Exception as e:
            logger.error(f"Connection check failed: {e}")
            return False
    
    def get_account(self) -> Dict:
        """
        Get account information
        
        Returns:
            Dict: Account information
        """
        try:
            response = requests.get(
                f"{self.base_url}/v2/account",
                headers=self.headers
            )
            
            if response.status_code == 200:
                return response.json()
            else:
                logger.error(f"Error getting account: {response.text}")
                return {}
        except Exception as e:
            logger.error(f"Error getting account: {e}")
            return {}
    
    def get_account_value(self) -> float:
        """Get total account value"""
        account = self.get_account()
        if account:
            return float(account.get('equity', 0))
        return 0.0
    
    def submit_order(self, symbol: str, side: str, quantity: float,
                     order_type: str = 'market', limit_price: float = None,
                     stop_price: float = None) -> Optional[Dict]:
        """
        Submit an order to Alpaca
        
        Args:
            symbol: Symbol to trade
            side: 'buy' or 'sell'
            quantity: Quantity to trade
            order_type: 'market', 'limit', 'stop', 'stop_limit'
            limit_price: Limit price for limit orders
            stop_price: Stop price for stop orders
        
        Returns:
            Dict: Order response or None if failed
        """
        if not self.connected:
            logger.error("Not connected to Alpaca")
            return None
        
        try:
            # Prepare order data
            order_data = {
                'symbol': symbol,
                'side': side.lower(),
                'type': order_type,
                'qty': str(quantity),
                'time_in_force': 'day'
            }
            
            if order_type == 'limit' and limit_price:
                order_data['limit_price'] = str(limit_price)
            
            if order_type in ['stop', 'stop_limit'] and stop_price:
                order_data['stop_price'] = str(stop_price)
            
            if order_type == 'stop_limit' and limit_price:
                order_data['limit_price'] = str(limit_price)
            
            # Submit order
            response = requests.post(
                f"{self.base_url}/v2/orders",
                headers=self.headers,
                json=order_data
            )
            
            if response.status_code == 200:
                order = response.json()
                logger.info(f"Order submitted: {order.get('id')}")
                return order
            else:
                logger.error(f"Error submitting order: {response.text}")
                return None
                
        except Exception as e:
            logger.error(f"Error submitting order: {e}")
            return None
    
    def get_orders(self, status: str = 'open') -> List[Dict]:
        """
        Get orders by status
        
        Args:
            status: 'open', 'closed', 'all'
        
        Returns:
            List[Dict]: List of orders
        """
        try:
            response = requests.get(
                f"{self.base_url}/v2/orders",
                headers=self.headers,
                params={'status': status}
            )
            
            if response.status_code == 200:
                return response.json()
            else:
                logger.error(f"Error getting orders: {response.text}")
                return []
        except Exception as e:
            logger.error(f"Error getting orders: {e}")
            return []
    
    def cancel_order(self, order_id: str) -> bool:
        """
        Cancel an order
        
        Args:
            order_id: Order ID to cancel
        
        Returns:
            bool: True if cancelled successfully
        """
        try:
            response = requests.delete(
                f"{self.base_url}/v2/orders/{order_id}",
                headers=self.headers
            )
            
            if response.status_code == 200 or response.status_code == 204:
                logger.info(f"Order cancelled: {order_id}")
                return True
            else:
                logger.error(f"Error cancelling order: {response.text}")
                return False
        except Exception as e:
            logger.error(f"Error cancelling order: {e}")
            return False
    
    def get_positions(self) -> List[Dict]:
        """Get all positions"""
        try:
            response = requests.get(
                f"{self.base_url}/v2/positions",
                headers=self.headers
            )
            
            if response.status_code == 200:
                return response.json()
            else:
                logger.error(f"Error getting positions: {response.text}")
                return []
        except Exception as e:
            logger.error(f"Error getting positions: {e}")
            return []
    
    def close_position(self, symbol: str) -> bool:
        """
        Close a position
        
        Args:
            symbol: Symbol to close
        
        Returns:
            bool: True if closed successfully
        """
        try:
            response = requests.delete(
                f"{self.base_url}/v2/positions/{symbol}",
                headers=self.headers
            )
            
            if response.status_code == 200 or response.status_code == 204:
                logger.info(f"Position closed: {symbol}")
                return True
            else:
                logger.error(f"Error closing position: {response.text}")
                return False
        except Exception as e:
            logger.error(f"Error closing position: {e}")
            return False
    
    def get_current_price(self, symbol: str) -> Optional[float]:
        """
        Get current price of a symbol
        
        Args:
            symbol: Symbol to get price for
        
        Returns:
            float: Current price or None if failed
        """
        try:
            # Using Alpaca data API
            response = requests.get(
                f"{self.data_url}/v2/stocks/{symbol}/trades/latest",
                headers=self.headers
            )
            
            if response.status_code == 200:
                data = response.json()
                return float(data.get('trade', {}).get('p', 0))
            else:
                logger.error(f"Error getting price: {response.text}")
                return None
        except Exception as e:
            logger.error(f"Error getting price: {e}")
            return None
    
    def get_bars(self, symbol: str, timeframe: str = '1D',
                 limit: int = 100) -> List[Dict]:
        """
        Get historical bars
        
        Args:
            symbol: Symbol to get bars for
            timeframe: Timeframe (e.g., '1Min', '5Min', '1D')
            limit: Number of bars to get
        
        Returns:
            List[Dict]: Historical bars
        """
        try:
            response = requests.get(
                f"{self.data_url}/v2/stocks/{symbol}/bars",
                headers=self.headers,
                params={
                    'timeframe': timeframe,
                    'limit': limit,
                    'adjustment': 'all'
                }
            )
            
            if response.status_code == 200:
                return response.json().get('bars', [])
            else:
                logger.error(f"Error getting bars: {response.text}")
                return []
        except Exception as e:
            logger.error(f"Error getting bars: {e}")
            return []

# Example usage
if __name__ == "__main__":
    bridge = AlpacaBridge()
    
    print("Alpaca Bridge Test")
    print("=" * 40)
    
    if bridge.connected:
        # Get account
        account = bridge.get_account()
        print(f"Account Equity: ${account.get('equity', 0)}")
        
        # Get price
        price = bridge.get_current_price('BTCUSD')
        if price:
            print(f"BTCUSD Price: ${price:.2f}")
        
        # Get positions
        positions = bridge.get_positions()
        print(f"Positions: {len(positions)}")
    else:
        print("Not connected - check API keys")
